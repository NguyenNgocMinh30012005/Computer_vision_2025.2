\begin{frame}{Experiment Variables}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Controlled factors}
    \begin{itemize}
      \item Number of input views
      \item Scene-level train / validation / test split
      \item Fixed threshold on final test
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Observed outcomes}
    \begin{itemize}
      \item Reconstruction accuracy
      \item Completeness and consistency
      \item Runtime per scene
      \item Occlusion and repeated-structure failures
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Training Strategy}
  \footnotesize
  \begin{enumerate}
    \item First build a supervised label cache with per-view visibility,
    occlusion, floating/wrong-depth, and match labels.
    \item Freeze MV-DUSt3R+ and train small reliability / match heads on
    generated GT labels.
    \item Evaluate OARH on occlusion-heavy scenes and RSDH on repeated-structure
    scenes before mixing them into the full pipeline.
    \item If validation improves, unfreeze only the confidence head and last
    decoder blocks for light fine-tuning.
    \item Keep final test clean: no threshold tuning, no checkpoint selection
    on test scenes.
  \end{enumerate}
\end{frame}
