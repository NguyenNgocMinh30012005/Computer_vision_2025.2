\begin{frame}{Problem Statement}
\begin{block}{Research objective}
Improve sparse-view 3D reconstruction on unseen indoor scenes by reducing three common failure modes: far-reference views, severe occlusion, and repeated structures.
\end{block}

\begin{itemize}
    \item Sparse-view settings provide limited geometric evidence and weaker co-visibility across images.
    \item Naive fusion of all predicted 3D points often introduces noise, structural inconsistency, and unstable point clouds.
    \item The project targets a modular extension to a pretrained reconstruction backbone rather than a new backbone architecture.
\end{itemize}
\end{frame}

\begin{frame}{Baseline and Technical Gap}
\begin{columns}[T,totalwidth=\textwidth]
\column{0.58\textwidth}
\begin{block}{Baseline backbone}
\textbf{MV-DUSt3R+} is a single-stage multi-view reconstruction network designed for sparse-view scene reconstruction.
\end{block}

\textbf{Observed limitations in sparse-view scenes}
\begin{itemize}
    \item \textbf{Far-reference views}: large viewpoint gaps reduce reliable correspondence.
    \item \textbf{Occlusion}: unsupported regions may still contribute to the final point cloud.
    \item \textbf{Repeated structures}: visually similar regions increase matching ambiguity.
\end{itemize}

\vspace{0.3em}
\textbf{Fusion weight}
\[
W_{\mathrm{final}}(p)=C(p)\,(1-O(p))\,S_{mv}(p)\,(1-A(p))\,G(p)\,M(p)
\]
\vspace{-0.7em}
\begin{itemize}
    \item $C$: confidence, $O$: occlusion, $S_{mv}$: multi-view support
    \item $A$: ambiguity, $G$: anchor support, $M$: multi-view agreement
\end{itemize}
\end{frame}

\begin{frame}{Proposed Contributions}
\begin{enumerate}
    \item \textbf{Visibility-aware view selection}
    \begin{itemize}
        \item Compare random, coverage-aware, diversity-aware, and hybrid view policies.
        \item Reduce far-reference cases and improve co-visibility before inference.
    \end{itemize}

    \item \textbf{Occlusion-aware weighted fusion}
    \begin{itemize}
        \item Suppress pixels that are weakly supported, poorly reprojected, or likely occluded.
        \item Reduce unstable geometry in the final point cloud.
    \end{itemize}

    \item \textbf{Ambiguity-aware repeated-structure filtering}
    \begin{itemize}
        \item Use ambiguity estimation, anchor pixels, and multi-view agreement for repeated indoor patterns.
    \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Repeated-Structure Filtering Strategy}
\begin{columns}[T,totalwidth=\textwidth]
\column{0.57\textwidth}
\textbf{Ambiguity estimation}
\[
A(p)=\alpha S(p)+\beta U(p)+\gamma R(p)
\]
\begin{itemize}
    \item $S(p)$: self-similarity within the image
    \item $U(p)=1-C(p)$: uncertainty from model confidence
    \item $R(p)$: reprojection disagreement across views
\end{itemize}

\textbf{Decision principle}
A candidate region is preserved only when it satisfies:
\begin{itemize}
    \item low ambiguity,
    \item support from reliable anchors,
    \item agreement across multiple views.
\end{itemize}

\column{0.43\textwidth}
\begin{block}{Targeted failure case}
Repeated structures such as chairs, windows, and cabinet patterns may produce geometrically incorrect but visually plausible matches.
\end{block}

\begin{block}{Expected effect}
\begin{itemize}
    \item fewer false correspondences,
    \item cleaner repeated-pattern regions,
    \item improved structural consistency.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Dataset}

\textbf{Primary dataset: ScanNet++}

\begin{frame}{Experimental Protocol and Feasibility}
\textbf{Protocol}
\begin{itemize}
    \item Evaluate on unseen indoor scenes under 2-, 3-, and 5-view settings.
    \item Compare the baseline MV-DUSt3R+ with each proposed module and with the full pipeline.
\end{itemize}

\textbf{Evaluation criteria}
\begin{itemize}
    \item reconstruction accuracy, completeness, and consistency,
    \item runtime per scene,
    \item case-based analysis on far-reference, occlusion-heavy, and repeated-structure scenarios.
\end{itemize}

\textbf{Why the project is feasible}
\begin{itemize}
    \item The project builds on a pretrained backbone and introduces modular additions rather than a new backbone.
    \item Each module can be implemented and evaluated incrementally.
\end{itemize}
\end{frame}
